IsPossible <-
function(i,j) {
# if !Runner @ Home
p <- ((bitwAnd(i,1)==1) &&
#  next to plate
      (i+1 == j)) ||
# if Runner @ Home
     ((bitwAnd(i,1)==0) &&
#  !Reflexive
      (i != j) &&
#  !Enough Players involved
      (Outs[i]+Runners[i]>=Outs[j]+Runners[j]) &&
#  Outs cannot decrease
      (Outs[i] <= Outs[j]) &&
      (sum(Bases[i,1:1])>=sum(Bases[j,1:1])) &&
      (sum(Bases[i,1:2])>=sum(Bases[j,1:2])) &&
      (sum(Bases[i,1:3])>=sum(Bases[j,1:3])) &&
      (sum(Bases[i,1:4])>=sum(Bases[j,1:4])))
p
}
